    $table->string('linked_entity_type', 50)->nullable();
    $table->char('linked_entity_id', 26)->nullable();

    $table->timestamps();
    $table->softDeletes();

    $table->index(['account_id', 'month_key']);
});

## 2. Models

Все модели используют **ULID** как primary key:

```php
// Трейт для всех Led_* моделей
trait UsesUlid {
    protected static function boot() {
        parent::boot();
        static::creating(function ($model) {
            if (empty($model->id)) {
                $model->id = \Str::ulid();
            }
        });
    }
    public function getIncrementing() { return false; }
    public function getKeyType() { return 'string'; }
}
```

**Важно:** все флаги (`is_disabled`, `is_archived`, `is_pinned`) кастовать в `boolean`, суммы — в `integer`.

```php
// LedTransaction
protected $casts = [
    'amount'      => 'integer',
    'sort_order'  => 'integer',
    'is_disabled' => 'boolean',
    'is_pinned'   => 'boolean',
    'occurred_at' => 'date:Y-m-d',
];
```

## 4. Эндпоинты — детали

### GET /Ledger/accounts
Возвращает счета юзера + `balance_today`:

```php
public function index(Request $request) {
    $user = $request->user();
    $layer = LedLayer::where('user_id', $user->id)
                     ->where('type', 'base')->first();

    return LedAccount::where('user_id', $user->id)
        ->where('layer_id', $layer->id)
        ->where('is_archived', 0)
        ->orderBy('sort_order')
        ->get()
        ->map(function ($account) use ($layer) {
            $account->balance_today = $this->calcBalanceToday($account, $layer);
            return $account;
        });
}

private function calcBalanceToday(LedAccount $account, LedLayer $layer): int {
    $today    = now()->format('Y-m-d');
    $monthKey = now()->format('Y-m');
    $prevKey  = now()->subMonth()->format('Y-m');

    // opening = closing предыдущего месяца
    $prevTotal = LedMonthTotal::where('account_id', $account->id)
        ->where('layer_id', $layer->id)
        ->where('month_key', $prevKey)
        ->first();

    $opening = $prevTotal?->closing_balance ?? $account->opening_balance;

    // delta = сумма транзакций с начала месяца по сегодня
    $delta = LedTransaction::where('account_id', $account->id)
        ->where('layer_id', $layer->id)
        ->where('is_disabled', 0)
        ->whereNull('deleted_at')
        ->whereMonth('occurred_at', now()->month)
        ->whereYear('occurred_at', now()->year)
        ->where('occurred_at', '<=', $today)
        ->get()
        ->sum(fn($tx) => in_array($tx->flow_kind, ['income','transfer_in'])
            ? $tx->amount : -$tx->amount
        );

    return $opening + $delta;
}
```

### GET /Ledger/transactions
Параметры: `start`, `end`, `account_id` (может быть comma-separated для нескольких счетов):

```php
public function index(Request $request) {
    $user  = $request->user();
    $query = LedTransaction::where('user_id', $user->id)
        ->whereNull('deleted_at')
        ->whereBetween('occurred_at', [
            $request->get('start'),
            $request->get('end'),
        ])
        ->orderBy('occurred_at', 'desc')
        ->orderBy('sort_order');

    // Фильтр по счетам — comma-separated
    if ($request->filled('account_id')) {
        $ids = explode(',', $request->get('account_id'));
        $query->whereIn('account_id', $ids);
    }

    return $query->with('tags')->get();
}
```

### POST /Ledger/transactions
```php
// Перевод = 2 транзакции атомарно
public function store(Request $request) {
    $data = $request->validate([
        'account_id'        => 'required|string',
        'target_account_id' => 'nullable|string',
        'flow_kind'         => 'required|in:expense,income,transfer_out,transfer_in,adjustment',
        'amount'            => 'required|integer|min:1', // копейки, > 0
        'occurred_at'       => 'required|date',
        'title'             => 'nullable|string|max:255',
        'note'              => 'nullable|string',
        'status'            => 'nullable|in:cleared,pending',
        'group_id'          => 'nullable|string',
    ]);

    $user    = $request->user();
    $layer   = LedLayer::where('user_id', $user->id)->where('type', 'base')->first();
    $monthKey = \Carbon\Carbon::parse($data['occurred_at'])->format('Y-m');

    DB::transaction(function () use ($data, $user, $layer, $monthKey) {
        $tx = LedTransaction::create([
            ...$data,
            'user_id'   => $user->id,
            'layer_id'  => $layer->id,
            'month_key' => $monthKey,
        ]);

        // Если перевод — создаём парную транзакцию
        if ($data['flow_kind'] === 'transfer_out' && $data['target_account_id']) {
            LedTransaction::create([
                'user_id'           => $user->id,
                'layer_id'          => $layer->id,
                'account_id'        => $data['target_account_id'],
                'target_account_id' => $data['account_id'],
                'flow_kind'         => 'transfer_in',
                'amount'            => $data['amount'],
                'occurred_at'       => $data['occurred_at'],
                'month_key'         => $monthKey,
                'title'             => $data['title'] ?? null,
                'status'            => $data['status'] ?? 'cleared',
                // Связываем пару
                'original_transaction_id' => $tx->id,
            ]);
        }

        // Пометить month_totals как dirty
        $this->markDirty($user->id, $layer->id, $data['account_id'], $monthKey);
        if ($data['target_account_id'] ?? null) {
            $this->markDirty($user->id, $layer->id, $data['target_account_id'], $monthKey);
        }
    });
}
```

### PATCH /Ledger/transactions/{id}/move
DnD — изменить дату и/или счёт:

```php
public function move(Request $request, string $id) {
    $data = $request->validate([
        'occurred_at' => 'nullable|date',
        'account_id'  => 'nullable|string',
    ]);

    $tx   = LedTransaction::where('user_id', $request->user()->id)->findOrFail($id);
    $old  = ['account_id' => $tx->account_id, 'month_key' => $tx->month_key];

    if ($data['occurred_at'] ?? null) {
        $tx->occurred_at = $data['occurred_at'];
        $tx->month_key   = \Carbon\Carbon::parse($data['occurred_at'])->format('Y-m');
    }
    if ($data['account_id'] ?? null) {
        $tx->account_id = $data['account_id'];
    }
    $tx->save();

    // Пометить dirty оба месяца и оба счёта
    $layer = LedLayer::where('user_id', $request->user()->id)->where('type', 'base')->first();
    $this->markDirty($request->user()->id, $layer->id, $old['account_id'], $old['month_key']);
    $this->markDirty($request->user()->id, $layer->id, $tx->account_id, $tx->month_key);

    return $tx;
}
```

public function toggle(Request $request, string $id) {
    $group = LedTransactionGroup::where('user_id', $request->user()->id)->findOrFail($id);
    $group->is_disabled = $request->boolean('is_disabled');
    $group->save();

    // Синхронно обновляем is_disabled на всех транзакциях группы
    LedTransaction::where('group_id', $group->id)
        ->update(['is_disabled' => $group->is_disabled]);

    return $group;
}

## 5. Пересчёт month_totals

```php
// Вызывается после любого изменения транзакций
private function markDirty(string $userId, string $layerId, string $accountId, string $monthKey): void {
    LedMonthTotal::updateOrCreate(
        ['layer_id' => $layerId, 'account_id' => $accountId, 'month_key' => $monthKey],
        ['user_id' => $userId, 'is_dirty' => 1, 'updated_at' => now()]
    );
}

// Пересчёт одного месяца для одного счёта
public function recalcMonth(string $layerId, string $accountId, string $monthKey): void {
    // opening = closing предыдущего месяца
    $prev = \Carbon\Carbon::createFromFormat('Y-m', $monthKey)->subMonth()->format('Y-m');
    $prevTotal = LedMonthTotal::where(['layer_id' => $layerId, 'account_id' => $accountId, 'month_key' => $prev])->first();
    $opening = $prevTotal?->closing_balance ?? 0;

    $txs = LedTransaction::where('account_id', $accountId)
        ->where('layer_id', $layerId)
        ->where('month_key', $monthKey)
        ->where('is_disabled', 0)
        ->whereNull('deleted_at')
        ->get();

    $income      = $txs->where('flow_kind', 'income')->sum('amount');
    $expense     = $txs->where('flow_kind', 'expense')->sum('amount');
    $transferIn  = $txs->where('flow_kind', 'transfer_in')->sum('amount');
    $transferOut = $txs->where('flow_kind', 'transfer_out')->sum('amount');
    $adjustment  = $txs->whereIn('flow_kind', ['adjustment'])->sum('amount');

    $closing = $opening + $income - $expense + $transferIn - $transferOut + $adjustment;

    LedMonthTotal::updateOrCreate(
        ['layer_id' => $layerId, 'account_id' => $accountId, 'month_key' => $monthKey],
        [
            'opening_balance'    => $opening,
            'closing_balance'    => $closing,
            'income_total'       => $income,
            'expense_total'      => $expense,
            'transfer_in_total'  => $transferIn,
            'transfer_out_total' => $transferOut,
            'adjustment_total'   => $adjustment,
            'tx_count'           => $txs->count(),
            'is_dirty'           => 0,
            'updated_at'         => now(),
        ]
    );
}
```
