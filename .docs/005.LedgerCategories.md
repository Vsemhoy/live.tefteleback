## Миграции

### 1. `create_Led_categories_table`

```php
Schema::create('Led_categories', function (Blueprint $table) {
    $table->char('id', 26)->primary();
    $table->char('user_id', 26)->index();
    $table->char('parent_id', 26)->nullable()->index();
    $table->string('name', 100);
    $table->tinyInteger('depth')->default(0);      // 0..4
    $table->string('path', 200)->nullable();       // '01K.01K.01K' — materialized path
    $table->integer('sort_order')->default(0);
    $table->tinyInteger('is_archived')->default(0);
    $table->timestamps();
});
```

### 2. `add_category_id_to_Led_transactions`

```php
Schema::table('Led_transactions', function (Blueprint $table) {
    $table->char('category_id', 26)->nullable()->after('group_id')->index();
});
```

## Контроллер `LedgerCategoryController`

### `index`
```php
// GET /Ledger/categories
// Отдаём плоский список — фронт сам строит дерево
public function index(Request $request)
{
    $categories = LedCategory::where('user_id', auth()->id())
        ->orderBy('path')   // materialized path = правильный порядок дерева
        ->get();

    return response()->json(['status' => 1, 'content' => $categories]);
}
```

### `store`
```php
// POST /Ledger/categories
public function store(Request $request)
{
    $data = $request->validate([
        'name'      => 'required|string|max:100',
        'parent_id' => 'nullable|string|size:26',
        'depth'     => 'integer|min:0|max:4',
        'sort_order'=> 'integer',
    ]);

    // Проверяем depth <= 4
    if (($data['depth'] ?? 0) > 4) {
        return response()->json(['message' => 'Max depth is 5 levels'], 422);
    }

    $id = (string) \Ulid\Ulid::generate();

    // Строим path: берём path родителя + текущий id
    $path = $id;
    if (!empty($data['parent_id'])) {
        $parent = LedCategory::where('id', $data['parent_id'])
                              ->where('user_id', auth()->id())
                              ->firstOrFail();
        $path = $parent->path . '.' . $id;
    }

    $category = LedCategory::create([
        'id'         => $id,
        'user_id'    => auth()->id(),
        'parent_id'  => $data['parent_id'] ?? null,
        'name'       => $data['name'],
        'depth'      => $data['depth'] ?? 0,
        'path'       => $path,
        'sort_order' => $data['sort_order'] ?? 0,
    ]);

    return response()->json(['status' => 1, 'content' => $category]);
}
```

### `update`
```php
// PUT /Ledger/categories/{id}
public function update(Request $request, string $id)
{
    $category = LedCategory::where('id', $id)
                            ->where('user_id', auth()->id())
                            ->firstOrFail();

    $data = $request->validate([
        'name'       => 'sometimes|string|max:100',
        'sort_order' => 'sometimes|integer',
        'is_archived'=> 'sometimes|boolean',
    ]);

    // parent_id и depth не меняем через update — только через отдельный move если понадобится
    $category->update($data);

    return response()->json(['status' => 1, 'content' => $category]);
}
```

### `destroy`
```php
// DELETE /Ledger/categories/{id}
public function destroy(string $id)
{
    $category = LedCategory::where('id', $id)
                            ->where('user_id', auth()->id())
                            ->firstOrFail();

    // Блокируем удаление если есть дети
    if ($category->children()->exists()) {
        return response()->json(['message' => 'Remove child categories first'], 422);
    }

    // Обнуляем category_id у транзакций
    LedTransaction::where('category_id', $id)->update(['category_id' => null]);

    $category->delete();

    return response()->json(['status' => 1]);
}
```

### `reorder`
```php
// POST /Ledger/categories/reorder
public function reorder(Request $request)
{
    $items = $request->validate([
        'items'            => 'required|array',
        'items.*.id'       => 'required|string',
        'items.*.sort_order'=> 'required|integer',
    ])['items'];

    foreach ($items as $item) {
        LedCategory::where('id', $item['id'])
                   ->where('user_id', auth()->id())
                   ->update(['sort_order' => $item['sort_order']]);
    }

    return response()->json(['status' => 1]);
}
```
